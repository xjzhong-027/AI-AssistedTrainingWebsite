@login_required def all_pages_status(request, exam_id): exam = get_object_or_404(Exam, id=exam_id) student_exam_record = get_object_or_404(StudentExamRecord, user=request.user, exam=exam) pages = exam.pages.all() pages_status = [] for page in pages: page_record, created = StudentPageRecord.objects.get_or_create(student_exam_record=student_exam_record, page=page) pages_status.append({ 'page_number': page.page_number, 'submitted': page_record.submitted, 'can_modify': page.can_modify, 'can_jump': page_record.submitted or page.page_number == 1 # 只有提交了的页面或第一页可以跳转 }) return JsonResponse({'status': 'success', 'pages': pages_status})
